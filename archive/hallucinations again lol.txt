1. mock\_data\_generator.py:
```python
import random
import string
from datetime import datetime, timedelta

# Placeholder: Use real-world data ranges and distributions for simulation
TEMP_MIN = -2  # °C
TEMP_MAX = 40  # °C
SALINITY_MIN = 0  # PSU
SALINITY_MAX = 40  # PSU
ACOUSTIC_PING_ID_LENGTH = 10

# Constants for generating random data
SPECIES_LIST = ["dolphin", "whale", "shark", "tuna", "swordfish"]
LOCATIONS_LIST = ["North Atlantic", "South Pacific", "Indian Ocean"]

def generate_random_species():
    return random.choice(SPECIES_LIST)

def generate_random_location():
    return random.choice(LOCATIONS_LIST)

def generate_acoustic_ping_id():
    chars = string.ascii_uppercase + string.digits
    return ''.join(random.choice(chars) for _ in range(ACOUSTIC_PING_ID_LENGTH))

def simulate_data(num_records):
    for _ in range(num_records):
        temp = random.uniform(TEMP_MIN, TEMP_MAX)
        salinity = random.uniform(SALINITY_MIN, SALINITY_MAX)
        acoustic_ping_id = generate_acoustic_ping_id()
        species = generate_random_species()
        location = generate_random_location()
        timestamp = datetime.now() - timedelta(days=random.randint(0, 365))

        print(f"Temperature: {temp}, Salinity: {salinity}, Acoustic Ping: {acoustic_ping_id}, Species: {species}, Location: {location}, Timestamp: {timestamp}")

if __name__ == "__main__":
    simulate_data(10)
```
